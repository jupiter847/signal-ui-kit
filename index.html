your-project / index.html
<!-- A small starting point -->
<link rel="stylesheet" href="./signal-ui.css">
<button id="open">Open menu</button>

<script type="module">
  import { createCommandMenu }
    from './signal-ui.js';

  const menu = createCommandMenu({
    trigger: document.querySelector('#open'),
    commands: [{
      id: 'hello',
      label: 'Say hello',
      onSelect: () => console.log('Hello!')
    }]
  });
</script>
